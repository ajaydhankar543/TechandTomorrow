sudo nano /etc/nginx/sites-available/techandtomorrow.social





server {
    listen 80;
    server_name techandtomorrow.social www.techandtomorrow.social;
    return 301 https://$host$request_uri;
}
server {
    listen 443 ssl;
    server_name techandtomorrow.social www.techandtomorrow.social;

    ssl_certificate /etc/letsencrypt/live/techandtomorrow.social/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/techandtomorrow.social/privkey.pem;

    root /home/ubuntu/TechandTomorrow/Frontend/dist;
    index index.html index.htm;

    location / {
        try_files $uri $uri/ =404;
    }

location /api {
    proxy_pass http://localhost:3000;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection 'upgrade';
    proxy_set_header Host $host;
    proxy_cache_bypass $http_upgrade;
}
}





sudo nginx -t && sudo systemctl restart nginx
